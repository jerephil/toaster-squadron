cmake_minimum_required(VERSION 3.24)
project(ToasterSquadron VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Build with MSVC and the Windows SDK on Windows (see GitHub Actions).")
endif()
enable_language(RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(ToasterSquadron WIN32
    src/main.cpp src/settings.cpp src/renderer.cpp resources/toaster.rc)
set_target_properties(ToasterSquadron PROPERTIES SUFFIX ".scr")
target_include_directories(ToasterSquadron PRIVATE src resources "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(ToasterSquadron PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(ToasterSquadron PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_link_libraries(ToasterSquadron PRIVATE d2d1 d3d11 dxgi windowscodecs ole32 comctl32 shell32 advapi32 user32 gdi32)
target_link_options(ToasterSquadron PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
include(CTest)
if(BUILD_TESTING)
    add_executable(contract_tests tests/contract_tests.cpp)
    target_include_directories(contract_tests PRIVATE src)
    target_compile_definitions(contract_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_compile_options(contract_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
    target_link_libraries(contract_tests PRIVATE user32)
    add_test(NAME screensaver_contract COMMAND contract_tests $<TARGET_FILE:ToasterSquadron>)
    set_tests_properties(screensaver_contract PROPERTIES TIMEOUT 45)
endif()
